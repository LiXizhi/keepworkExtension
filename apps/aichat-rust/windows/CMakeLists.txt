cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(keepwork_windows_bootstrap LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(INSTALL_PROJECT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz URL https://github.com/richgel999/miniz/archive/refs/tags/3.0.2.tar.gz
  URL_HASH SHA256=c4b4c25a4eb81883448ff8924e6dba95c800094a198dc9ce66a292ac2ef8e018)
FetchContent_MakeAvailable(json miniz)
add_library(keepwork_policy common/policy.cpp common/archive.cpp)
target_include_directories(keepwork_policy PUBLIC common)
target_link_libraries(keepwork_policy PUBLIC nlohmann_json::nlohmann_json miniz)
include(CTest)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Build the Windows launcher with an MSVC-compatible toolchain")
  endif()
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  set(KEEPWORK_PUBLIC_KEY_HEADER "" CACHE FILEPATH "Generated pinned RSA public key header")
  if(NOT EXISTS "${KEEPWORK_PUBLIC_KEY_HEADER}")
    message(FATAL_ERROR "Generate the launcher trust root with scripts/cef-release.cjs key-header")
  endif()
  configure_file("${KEEPWORK_PUBLIC_KEY_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/release-key.hpp" COPYONLY)
  FetchContent_Declare(webview URL https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/1.0.2903.40
    DOWNLOAD_NAME webview2.zip
    URL_HASH SHA256=ef128016dd1e51c59178c827ed5b8aa3322c57afa8675d930f8109505542ad74)
  FetchContent_MakeAvailable(webview)
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    set(KW_ARCH arm64)
  elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(KW_ARCH x64)
  else()
    set(KW_ARCH x86)
  endif()
  add_executable(Keepwork WIN32 launcher.cpp native.cpp)
  target_include_directories(Keepwork PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${webview_SOURCE_DIR}/build/native/include")
  target_link_libraries(Keepwork PRIVATE keepwork_policy winhttp bcrypt crypt32 ole32 shell32 user32 advapi32 comctl32)
  target_compile_options(Keepwork PRIVATE /utf-8 /W4)
  target_compile_options(keepwork_policy PRIVATE /utf-8)
  if(BUILD_TESTING)
    add_executable(keepwork_native_verify tests/native_verify.cpp native.cpp)
    target_include_directories(keepwork_native_verify PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
      "${CMAKE_CURRENT_BINARY_DIR}" "${webview_SOURCE_DIR}/build/native/include")
    target_link_libraries(keepwork_native_verify PRIVATE keepwork_policy winhttp bcrypt crypt32 ole32 shell32 user32 advapi32 comctl32)
    target_compile_options(keepwork_native_verify PRIVATE /utf-8 /W4)
  endif()
  if(KW_ARCH STREQUAL "arm64")
    target_link_options(Keepwork PRIVATE /SUBSYSTEM:WINDOWS,10.00)
  else()
    target_link_options(Keepwork PRIVATE /SUBSYSTEM:WINDOWS,6.01)
  endif()
  add_custom_command(TARGET Keepwork POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${webview_SOURCE_DIR}/build/native/${KW_ARCH}/WebView2Loader.dll" "$<TARGET_FILE_DIR:Keepwork>/WebView2Loader.dll")
  add_custom_command(TARGET Keepwork POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:Keepwork>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${webview_SOURCE_DIR}/LICENSE.txt" "$<TARGET_FILE_DIR:Keepwork>/licenses/WebView2-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${webview_SOURCE_DIR}/NOTICE.txt" "$<TARGET_FILE_DIR:Keepwork>/licenses/WebView2-NOTICE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${json_SOURCE_DIR}/LICENSE.MIT" "$<TARGET_FILE_DIR:Keepwork>/licenses/nlohmann-json-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${miniz_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:Keepwork>/licenses/miniz-LICENSE.txt")
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/policy_test.cpp")
  add_executable(keepwork_policy_test tests/policy_test.cpp)
  target_link_libraries(keepwork_policy_test PRIVATE keepwork_policy)
  add_test(NAME keepwork_policy COMMAND keepwork_policy_test)
endif()
